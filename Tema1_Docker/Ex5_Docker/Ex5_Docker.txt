#Dorin: Am creat scriptul Ex5_file_mover.py

	import os
	import shutil
	import time
	import random

	source='/input'
	destination='/output'

	while True:
	    fisiere=sorted(os.listdir(source))
	    cale_completa_source=source+'/'+fisiere[0]
	    cale_completa_destination=destination+'/'+fisiere[0]
	    shutil.move(cale_completa_source, cale_completa_destination)
	    time.sleep(random.randint(2,5))

vim Dockerfile.Ex5

	FROM python:slim

	COPY Ex5_file_mover.py .

	CMD ["python", "Ex5_file_mover.py"]

docker build -f Dockerfile.Ex5 -t militarueduarddorin/ex5-file-mover .

docker run --rm -v /tmp/input:/input -v /tmp/output:/output --name Ex5_File_Mover militarueduarddorin/ex5-file-mover:latest
